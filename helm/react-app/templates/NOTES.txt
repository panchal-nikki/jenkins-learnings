{{- if eq .Values.service.type "NodePort" }}
NodePort access (Minikube):
  minikube service {{ include "react-app.fullname" . }}-service -n {{ .Values.namespace }}

Or manually:
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "http://$NODE_IP:{{ .Values.service.nodePort }}"
{{- end }}

{{- if .Values.ingress.enabled }}

Ingress access:
  1. Add this to your /etc/hosts (or C:\Windows\System32\drivers\etc\hosts on Windows):
       127.0.0.1  {{ .Values.ingress.host }}
  2. Open: http://{{ .Values.ingress.host }}
{{- end }}
